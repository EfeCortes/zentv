        ];

        // --- CITAS MULTILINGÜES ---
        var allQuotes = [
            { es: "La simplicidad es la máxima sofisticación.", en: "Simplicity is the ultimate sophistication.", fr: "La simplicité est la sophistication suprême.", de: "Einfachheit ist die höchste Stufe der Vollendung.", author: "Leonardo da Vinci" },

            { es: "Sólo sé que no sé nada.", en: "I only know that I know nothing.", fr: "Tout ce que je sais, c'est que je ne sais rien.", de: "Ich weiß, dass ich nichts weiß.", author: "Sócrates" },
            { es: "Donde hay amor hay vida.", en: "Where there is love there is life.", fr: "Là où il y a de l'amour, il y a de la vie.", de: "Wo Liebe ist, da ist auch Leben.", author: "Mahatma Gandhi" }

        var dailyQuotes = [];
        var activeGallery = [];
        var newsHeadlines = [];
        var currentSlide = 0;
        var quoteIndex = 0;
        var langIndex = 0; // 0: ES, 1: EN, 2: FR, 3: DE
        var languages = ['es', 'en', 'fr', 'de'];
        var newsIndex = 0;
        var lastDayProcessed = "";

        // --- MOTORES DE DATOS ---
        function fetchWeather() {
            var url = "https://api.open-meteo.com/v1/forecast?latitude=-17.3895&longitude=-66.1568&current=temperature_2m,weather_code&timezone=auto";
            fetch(url).then(function(res) { return res.json(); }).then(function(data) {
                if (data.current) {
                    document.getElementById('clima-temp').innerText = Math.round(data.current.temperature_2m) + "°";
                    document.getElementById('clima-desc').innerHTML = "COCHABAMBA<br>" + (data.current.weather_code <= 3 ? "DESPEJADO" : "NUBOSIDAD");
                }
            });
        }

        function fetchAllNews() {
            var sources = [
                { name: "BBC Mundo", url: "https://feeds.bbci.co.uk/mundo/rss.xml" },
                { name: "CNN en Español", url: "https://cnnespanol.cnn.com/feed/" },
                { name: "Los Tiempos", url: "https://www.lostiempos.com/rss/portada" },
                { name: "El Deber", url: "https://eldeber.com.bo/rss/portada" }
            ];
            
            // ANTI-CACHE AGRESIVO: _t=ms para obligar a la TV a descargar info nueva
            var promises = sources.map(function(src) {
                var apiUrl = "https://api.rss2json.com/v1/api.json?rss_url=" + encodeURIComponent(src.url) + "&count=10&_t=" + Date.now();
                return fetch(apiUrl).then(function(r){ return r.json(); }).then(function(data){
                    if(data.status === 'ok' && data.items) {
                        return data.items.map(function(item) {
                            return { title: item.title.trim(), source: src.name, date: formatRSSDate(item.pubDate) };
                        });
                    }
                    return [];
                }).catch(function(){ return []; });
            });

            Promise.all(promises).then(function(results) {
                var merged = [].concat.apply([], results);
                if (merged.length > 0) {
                    newsHeadlines = merged.sort(function() { return 0.5 - Math.random(); });
                }
            });
        }

        function formatRSSDate(dateStr) {
            if (!dateStr) return "";
            try {
                var d = new Date(dateStr);
                return (d.toLocaleDateString('es-ES', { day: '2-digit', month: 'short' }) + " • " + d.toLocaleTimeString('es-ES', { hour: '2-digit', minute: '2-digit', hour12: false })).toUpperCase();
            } catch (e) { return "RECIENTE"; }
        }

        // --- LÓGICA DE INTERFAZ ---
        function initDailyContent() {
            var now = new Date();
            var dayString = now.toDateString();
            
            dailyQuotes = allQuotes.slice().sort(function() { return 0.5 - Math.random(); });
            
            if (lastDayProcessed === dayString) return;

            currentSlide = 0;
            newsIndex = 0;
            activeGallery = galleryData.slice().sort(function() { return 0.5 - Math.random(); });
            
            var container = document.getElementById('gallery-container');
            container.innerHTML = "";
            activeGallery.forEach(function(item, i) {
                var div = document.createElement('div');
                div.className = "slide" + (i === 0 ? " slide-active" : "");
                div.style.backgroundImage = "url('" + item.url + "')";
                container.appendChild(div);
            });

            setTimeout(function(){ 
                document.getElementById('loader').style.opacity = "0"; 
                setTimeout(function(){ document.getElementById('loader').style.display = "none"; }, 1000);
                document.getElementById('main-layout').style.opacity = "1";
                updateCaption();
                rotateQuotes(); 
                rotateNews();
            }, 1000);

            lastDayProcessed = dayString;
        }

        function cycleGallery() {
            var slides = document.querySelectorAll('.slide');
            if(slides.length === 0) return;
            for(var i=0; i<slides.length; i++) { slides[i].classList.remove('slide-previous'); }
            slides[currentSlide].classList.add('slide-previous');
            slides[currentSlide].classList.remove('slide-active');
            currentSlide = (currentSlide + 1) % slides.length;
            slides[currentSlide].classList.add('slide-active');
            updateCaption();
        }

        // ROTACIÓN DE CITAS (CADA 25 SEG - CAMBIO DE IDIOMA)
        function rotateQuotes() {
            var display = document.getElementById('quote-display');
            var authorEl = document.getElementById('quote-author');
            if (dailyQuotes.length === 0) return;
            
            display.style.opacity = 0; authorEl.style.opacity = 0;
            setTimeout(function() {
                var currentQuote = dailyQuotes[quoteIndex];
                var currentLangKey = languages[langIndex];
                
                display.innerText = currentQuote[currentLangKey];
                authorEl.innerText = currentQuote.author.toUpperCase();
                
                display.style.opacity = 1; authorEl.style.opacity = 1;

                // Avanzar al siguiente idioma, y si ya terminamos los 4, pasar a la siguiente cita
                langIndex++;
                if(langIndex >= languages.length) {
                    langIndex = 0;
                    quoteIndex = (quoteIndex + 1) % dailyQuotes.length;
                }
            }, 1000);
        }

        function rotateNews() {
            var nDisp = document.getElementById('noticia-display');
            var nFuente = document.getElementById('noticia-fuente');
            if (newsHeadlines.length === 0) return;
            
            nDisp.style.opacity = 0; nFuente.style.opacity = 0;
            setTimeout(function() {
                var item = newsHeadlines[newsIndex];
                if (item) {
                    nDisp.children[0].innerText = item.date;
                    nDisp.children[1].innerText = item.title;
                    nFuente.innerText = item.source.toUpperCase();
                    nDisp.style.opacity = 1; nFuente.style.opacity = 1;
                    newsIndex = (newsIndex + 1) % newsHeadlines.length;
                }
            }, 1000);
        }

        function updateCaption() {
            var infoEl = document.getElementById('info-obra');
            var currentArt = activeGallery[currentSlide];
            infoEl.style.opacity = 0;
            setTimeout(function() {
                if(currentArt) {
                    infoEl.innerText = currentArt.title + " • " + currentArt.author;
                    infoEl.style.opacity = 0.5;
                }
            }, 1000);
        }

        function updateClock() {
            var now = new Date();
            document.getElementById('reloj').innerText = now.toLocaleTimeString('es-BO', {hour: '2-digit', minute:'2-digit', hour12: false});
            document.getElementById('fecha-texto').innerText = now.toLocaleDateString('es-ES', { weekday: 'short', day: 'numeric', month: 'short' }).toUpperCase();
            if (now.toDateString() !== lastDayProcessed) { initDailyContent(); fetchAllNews(); }
        }

        window.onload = function() {
            initDailyContent();
            updateClock();
            fetchWeather();
            fetchAllNews();
            
            setInterval(updateClock, 1000);
            setInterval(cycleGallery, 45000); 
            
            // CAMBIOS CONTINUOS
            setInterval(rotateQuotes, 25000); // Cambio de IDIOMA cada 25 seg
            setInterval(rotateNews, 30000);   // Noticias cada 30 seg
            
            // REFRESH DE FONDO
            setInterval(fetchWeather, 20 * 60 * 1000);
            setInterval(fetchAllNews, 10 * 60 * 1000); // Sincronización real cada 10 min
        };
